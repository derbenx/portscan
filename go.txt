package main

import (
	"fmt"
	"net"
	"sync"
	"time"
)

// Result holds the outcome of a single port check
type Result struct {
	Port   int
	Open   bool
}

func scanPort(ip string, port int, timeout time.Duration, results chan<- Result, wg *sync.WaitGroup) {
	defer wg.Done()

	address := fmt.Sprintf("%s:%d", ip, port)
	conn, err := net.DialTimeout("tcp", address, timeout) // [1]

	if err != nil {
		results <- Result{Port: port, Open: false}
		return
	}
	conn.Close() // [2]
	results <- Result{Port: port, Open: true}
}

func main() {
	targetIP := "192.168.1.1" // Base IP from your code
	startPort := 1
	endPort := 1024
	timeout := 500 * time.Millisecond // Your 'to' value

	results := make(chan Result, 100)
	var wg sync.WaitGroup

	// Start a worker goroutine to process results as they come in
	go func() {
		for res := range results {
			if res.Open {
				fmt.Printf("[+] Port %d is OPEN\n", res.Port)
			}
		}
	}()

	fmt.Printf("Scanning %s...\n", targetIP)

	// Launch scans concurrently
	for port := startPort; port <= endPort; port++ {
		wg.Add(1)
		go scanPort(targetIP, port, timeout, results, &wg) // [3]
	}

	wg.Wait()
	close(results)
	fmt.Println("Scan complete.")
}

// [1] net.DialTimeout implements the core scanning logic efficiently.
// [2] Ensure connections are closed immediately to prevent file descriptor leaks.
// [3] Goroutines allow hundreds of ports to be checked in parallel.
